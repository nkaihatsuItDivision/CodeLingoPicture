@Entity
public class Student {
    @ManyToOne(fetch = FetchType.LAZY)
    private Course course;
}

// コントローラーで学生リスト取得
List<Student> students = repository.findAll();
for (Student s : students) {
    s.getCourse().getName(); // 追加クエリ発生！
}
  • 🔥 1回の学生取得 + 100回のコース取得
  • ⚠️ LazyInitializationException
  • 🐌 データベース負荷大

✅
最適化後
クエリ実行回数: 1回
// 最適化されたコード
@EntityGraph(attributePaths = {"course"})
@Query("SELECT s FROM Student s")
List<Student> findAllWithCourse();

// または JOIN FETCH使用
@Query("SELECT s FROM Student s JOIN FETCH s.course")
List<Student> findAllWithCourseFetch();

// 一括取得で完了
List<Student> students = repository.findAllWithCourse();
  • ⚡ 1回のJOIN クエリで完了
  • ✨ 例外解決完了
