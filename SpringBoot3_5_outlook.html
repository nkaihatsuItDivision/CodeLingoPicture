@Transactional
public void bulkDeleteStudents(
    List<Long> studentIds,
    String deleteType,
    String deleteReason,
    String deletedBy) {

    for (Long studentId : studentIds) {
        if ("logical".equals(deleteType)) {
            logicalDeleteStudent(studentId,
                deleteReason, deletedBy);
        } else {
            physicalDeleteStudent(studentId,
                deleteReason, deletedBy);
        }
    }
}

function updateSelectionStatus() {

        '.student-checkbox:checked');
    const count = checked.length;

    document.getElementById('selectedCount')
        .textContent = count + '件選択中';

        'bulkDeleteBtn');
    deleteBtn.disabled = count === 0;
}

@Transactional(timeout = 300) // 5分タイムアウト
public BulkDeleteResult bulkDelete(
    BulkDeleteRequest request) {

    List<String> successIds = new ArrayList<>();
    List<String> failedIds = new ArrayList<>();

    for (Long id : request.getStudentIds()) {
        try {
            deleteStudent(id, request);
            successIds.add(id.toString());
        } catch (Exception e) {
            failedIds.add(id.toString());
            log.error("Delete failed for: " + id, e);
        }
    }

    return new BulkDeleteResult(successIds, failedIds);
}
