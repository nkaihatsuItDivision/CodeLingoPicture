<!-- search-advanced.html の核心構造 -->
<form th:action="@{/students/search/advanced}"
      th:object="${searchCriteria}" method="post">

    <!-- 基本条件セクション -->
    <div class="form-section">
        <input th:field="*{name}" class="form-control" />
        <input th:field="*{minAge}" type="number" />
        <input th:field="*{maxAge}" type="number" />
    </div>

    <!-- 日付範囲セクション -->
    <div class="form-section">
        <input th:field="*{startDate}" type="date" />
        <input th:field="*{endDate}" type="date" />
    </div>

    <!-- 複数選択セクション -->
    <div class="form-section">
        <div th:each="major : ${availableMajors}">
            <input th:field="*{selectedMajors}"
                   th:value="${major}" type="checkbox" />
        </div>
    </div>

    <!-- 演算子選択 -->
    <div class="operator-selection">
        <input th:field="*{searchOperator}"
               value="AND" type="radio" />
        <input th:field="*{searchOperator}"
               value="OR" type="radio" />
    </div>

    <button type="submit">高度検索実行</button>
</form>
