    Set ws = Worksheets("存在しない")

End Sub

✅ エラー処理あり
Sub WithErrorHandling()
    On Error GoTo ErrorHandler
    Dim ws As Worksheet
    Set ws = Worksheets("存在しない")
    Exit Sub
ErrorHandler:
    MsgBox "シートが見つかりません"
End Sub
✅ エラーをキャッチ
わかりやすいメッセージを表示
😊 マクロが安全に終了
📢 ユーザーに状況を通知
🛡️ データを保護
