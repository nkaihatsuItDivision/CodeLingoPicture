LocalDate・DateTimeFormatter活用による期間指定検索機能

日付範囲検索の実装構造

開始日検索条件

LocalDateを使用したタイプセーフな開始日指定検索

// 開始日検索条件の実装
if (criteria.getStartDate() != null) {
    Predicate startDatePredicate = criteriaBuilder
        .greaterThanOrEqualTo(
            root.get("createdDate"),
            criteria.getStartDate()
        );
    predicates.add(startDatePredicate);
}

終了日検索条件

終了日の翌日未満条件で正確な期間検索を実現

// 終了日検索条件の実装
if (criteria.getEndDate() != null) {
    // 終了日の翌日を取得（時刻を考慮）
    LocalDate nextDay = criteria.getEndDate()
        .plusDays(1);
    
    Predicate endDatePredicate = criteriaBuilder
        .lessThan(
            root.get("createdDate"),
            nextDay
        );
    predicates.add(endDatePredicate);
}

日付バリデーション
// 日付バリデーションの実装
@AssertTrue(message = "開始日は終了日以前である必要があります")
public boolean isValidDateRange() {
    if (startDate == null || endDate == null) {
        return true; // null の場合はバリデーション対象外
    }
    return !startDate.isAfter(endDate);
}

// JavaScript での即座バリデーション
function validateDateRange() {
    const startDate = document.getElementById('startDate').value;
    const endDate = document.getElementById('endDate').value;
    
    if (startDate && endDate && startDate > endDate) {
        alert('開始日は終了日以前を指定してください');
        return false;
    }
    return true;
}
