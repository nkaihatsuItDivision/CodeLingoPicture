SelectionManagerクラス設計
複数ページにわたる選択状態を管理する専用クラスの設計と実装
class SelectionManager {
    constructor() {
        // 全選択状態を管理（ページ横断）
        this.selectedIds = new Set();
        
        // 現在ページの学生ID一覧
        this.currentPageIds = new Set();
        
        // ページネーション情報
        this.currentPage = 1;
        this.totalPages = 1;
    }

    // 状態初期化メソッド
    initializePage(pageStudents) {
        this.currentPageIds.clear();
        pageStudents.forEach(student => {
            this.currentPageIds.add(student.id);
        });
    }
}

クラス設計技術

  • Set()による効率的ID管理
  • ページ情報とのデータ分離
  • 状態管理の構造化

選択状態の永続化
ページ切り替え時でも選択状態を保持する高度な状態管理機能
// 選択状態の更新と保持
updateSelectionState(studentId, isSelected) {
    if (isSelected) {
        this.selectedIds.add(studentId);
    } else {
        this.selectedIds.delete(studentId);
    }
    
    // UI状態の即座反映
    this.updatePageDisplay();
    this.updateBulkActionButtons();
}

// ページ表示時の状態復元
restorePageSelections() {
    this.currentPageIds.forEach(studentId => {
        const checkbox = document
            .querySelector(`[value="${studentId}"]`);
        if (checkbox) {
            checkbox.checked =
                this.selectedIds.has(studentId);
        }
    });
}

状態管理技術

  • Set()によるID追加・削除
  • ページ切り替え時の状態復元
  • リアルタイムUI更新

ページネーション連携
複数ページにわたる選択操作と全選択・部分選択の完全対応
// ページ切り替え時の処理
onPageChange(newPage, newPageStudents) {
    // 現在ページ情報更新
    this.currentPage = newPage;
    this.initializePage(newPageStudents);
    
    // 選択状態復元
    this.restorePageSelections();
    
    // 全選択状態の判定
    this.updateSelectAllState();
}

// 全選択状態の高度判定
updateSelectAllState() {
    const pageSelectedCount = Array.from(
        this.currentPageIds
    ).filter(id => this.selectedIds.has(id)).length;
    
    const selectAllCheckbox = document
        .getElementById('selectAll');
    
    selectAllCheckbox.checked =
        (pageSelectedCount === this.currentPageIds.size);
}

// 選択状態の総合監視
updateBulkActionButtons() {
    const totalSelected = this.selectedIds.size;
    const pageSelected = Array.from(
        this.currentPageIds
    ).filter(id => this.selectedIds.has(id)).length;
    
    // 一括削除ボタン制御
    const bulkDeleteBtn = document
        .getElementById('bulkDeleteBtn');
    
    if (totalSelected > 0) {
        bulkDeleteBtn.disabled = false;
        bulkDeleteBtn.textContent =
            `選択した${totalSelected}件を削除`;
    } else {
        bulkDeleteBtn.disabled = true;
        bulkDeleteBtn.textContent = '削除';
    }
    
    // 選択状況表示
    this.updateSelectionDisplay(totalSelected, pageSelected);
}

// 選択IDsリストの取得
getSelectedIds() {
    return Array.from(this.selectedIds);
}
