// リアルタイムバリデーション実装
document.addEventListener('DOMContentLoaded', function() {
    const nameInput = document.getElementById('name');
    const emailInput = document.getElementById('email');
    const gradeInput = document.getElementById('grade');

    // 学生名のリアルタイムチェック
    nameInput.addEventListener('input', function() {
        const value = this.value.trim();
        const feedback = this.nextElementSibling;

        if (value.length > 100) {
            showValidationError(this, '学生名は100文字以内で入力してください');
        } else if (value.length > 0) {
            showValidationSuccess(this, '入力形式が正しいです');
        } else {
            clearValidation(this);
        }
    });

    // 検索タイプ切り替え
    const searchTypeRadios = document.querySelectorAll('input[name="nameSearchType"]');
    searchTypeRadios.forEach(radio => {
        radio.addEventListener('change', function() {
            updateSearchTypePlaceholder(this.value);
            updateHelpText(this.value);
        });
    });

    // Ajax検索実装
    const searchForm = document.getElementById('searchForm');
    const ajaxSearchBtn = document.getElementById('ajaxSearch');

    ajaxSearchBtn.addEventListener('click', async function(e) {
        e.preventDefault();
        const formData = new FormData(searchForm);
        const searchData = Object.fromEntries(formData);

        try {
            showLoadingIndicator();
            const response = await fetch('/students/search', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(searchData)
            });

            const result = await response.json();
            updateSearchResults(result);
            hideLoadingIndicator();
        } catch (error) {
            console.error('Ajax検索エラー:', error);
            showErrorMessage('検索中にエラーが発生しました');
            hideLoadingIndicator();
        }
    });
});
