@GetMapping("/search/advanced")
public String advancedSearchForm(
    Model model,
    @ModelAttribute StudentSearchCriteria criteria) {

    // 選択肢データの取得
    List<String> availableMajors =
        studentSearchService.getAvailableMajors();
    List<String> availableStatuses =
        studentSearchService.getAvailableStatuses();

    model.addAttribute("availableMajors", availableMajors);
    model.addAttribute("availableStatuses", availableStatuses);
    model.addAttribute("searchCriteria", criteria);

    return "student/search-advanced";
}

@PostMapping("/search/advanced")
public String executeAdvancedSearch(
    @Valid @ModelAttribute StudentSearchCriteria criteria,
    BindingResult bindingResult,
    Model model,
    RedirectAttributes redirectAttributes) {

    // バリデーションエラーチェック
    if (bindingResult.hasErrors()) {
        // 選択肢データを再設定
        populateFormData(model);
        return "student/search-advanced";
    }

    // PRGパターン：検索条件をセッションに保存
    redirectAttributes.addFlashAttribute(
        "searchCriteria", criteria);

    return "redirect:/students/search/results";
}

@GetMapping("/api/search/preview")
@ResponseBody
public ResponseEntity<SearchPreviewResponse> getSearchPreview(
    @Valid StudentSearchCriteria criteria) {

    try {
        // 検索プレビューの実行
        long resultCount = studentSearchService
            .countAdvancedSearch(criteria);

        SearchPreviewResponse response =
            new SearchPreviewResponse(resultCount,
                criteria.getDisplayText());

        return ResponseEntity.ok(response);
    } catch (Exception e) {
        return ResponseEntity
            .badRequest().build();
    }
}

GET 検索結果表示

/students/search/results

Week 10.3拡張：高度検索結果の表示・ページネーション

@GetMapping("/search/results")
public String showSearchResults(
    @ModelAttribute("searchCriteria")
        StudentSearchCriteria criteria,
    @RequestParam(defaultValue = "0") int page,
    @RequestParam(defaultValue = "20") int size,
    Model model) {

    // Week 10.4 高度検索の実行
    Page<Student> students = studentSearchService
        .advancedSearchWithPaging(criteria, page, size);

    model.addAttribute("students", students);
    model.addAttribute("criteria", criteria);

    return "student/search-results";
}

// POST-REDIRECT-GET パターンの実装
@PostMapping("/search/advanced")
public String executeAdvancedSearch(
    @Valid @ModelAttribute StudentSearchCriteria criteria,
    BindingResult bindingResult,
    RedirectAttributes redirectAttributes) {

    if (bindingResult.hasErrors()) {
        // バリデーションエラー：再表示
        return "student/search-advanced";
    }

    // 検索条件をFlashAttributeで次の画面に渡す
    redirectAttributes.addFlashAttribute(
        "searchCriteria", criteria);

    // リダイレクトで重複送信防止
    return "redirect:/students/search/results";
}
