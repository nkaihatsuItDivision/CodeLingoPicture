StudentSearchController.java - Week 10.1/10.2統合実装
@Controller
@RequestMapping("/students")
@Slf4j
public class StudentSearchController {

    @Autowired
    private StudentSearchService studentSearchService; // Week 10.1基盤サービス

    @GetMapping("/search")
    public String showSearchForm(Model model) {
        model.addAttribute("searchForm", new StudentSearchForm());
        return "students/search";
    }

    @PostMapping("/search")
    public String executeSearch(@Valid @ModelAttribute("searchForm") StudentSearchForm searchForm,
                                BindingResult bindingResult,
                                RedirectAttributes redirectAttributes,
                                Model model) {

        log.info("検索リクエスト受信: {}", searchForm);

        // バリデーションエラーチェック
        if (bindingResult.hasErrors()) {
            log.warn("バリデーションエラー: {}", bindingResult.getAllErrors());
            model.addAttribute("searchForm", searchForm);
            return "students/search"; // エラー時は検索画面に戻る
        }

        try {
            // Week 10.1基盤サービス活用
            StudentSearchCriteria criteria = searchForm.toCriteria();
            List<StudentDto> searchResults = studentSearchService.searchStudents(criteria);

            log.info("検索完了: {}件のデータを取得", searchResults.size());

            // PRGパターン適用
            redirectAttributes.addFlashAttribute("searchResults", searchResults);
            redirectAttributes.addFlashAttribute("searchCriteria", criteria);
            redirectAttributes.addFlashAttribute("resultCount", searchResults.size());

            return "redirect:/students/search/results";

        } catch (Exception e) {
            log.error("検索処理中にエラーが発生", e);
            model.addAttribute("errorMessage", "検索処理中にエラーが発生しました");
            return "students/search";
        }
    }
}

Ajax対応・非同期処理サポート

@PostMapping(value = "/search", produces = "application/json")
@ResponseBody
public ResponseEntity<SearchResultResponse> executeSearchAjax(
        @Valid @RequestBody StudentSearchForm searchForm,
        BindingResult bindingResult) {

    if (bindingResult.hasErrors()) {
        return ResponseEntity.badRequest()
            .body(SearchResultResponse.error(bindingResult.getAllErrors()));
    }

    // Week 10.1サービス活用
    StudentSearchCriteria criteria = searchForm.toCriteria();
    List<StudentDto> results = studentSearchService.searchStudents(criteria);

    return ResponseEntity.ok(
        SearchResultResponse.success(results, criteria));
}

// JavaScript - Ajax検索実装
async function executeAjaxSearch() {
    const formData = new FormData(document.getElementById('searchForm'));
    const searchData = Object.fromEntries(formData);

    try {
        const response = await fetch('/students/search', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(searchData)
        });

        const result = await response.json();
        updateSearchResults(result);
    } catch (error) {
        console.error('検索エラー:', error);
    }
}
