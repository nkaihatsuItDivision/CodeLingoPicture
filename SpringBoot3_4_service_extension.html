advancedSearchWithPaging()

Page<Student> advancedSearchWithPaging(criteria, page, size)

複合条件・日付範囲・複数選択を統合した高度検索メソッド

@Service
@Transactional(readOnly = true)
public class StudentSearchService {

    @Autowired
    private StudentRepository studentRepository;

    /**
     * Week 10.4 高度検索メソッド
     */
    public Page<Student> advancedSearchWithPaging(
        StudentSearchCriteria criteria,
        int page, int size) {

        // ページネーション設定
        Pageable pageable = PageRequest.of(page, size,
            Sort.by("createdDate").descending());

        // 高度検索Specificationの作成
        Specification<Student> spec =
            StudentSpecifications.advancedSearch(criteria);

        // 検索実行
        return studentRepository.findAll(spec, pageable);
    }
}

getAvailableMajors()

List<String> getAvailableMajors()

データベースから専攻一覧を取得・キャッシュ機能付き

/**
 * 利用可能な専攻一覧を取得
 * @Cacheable によるキャッシュ機能
 */
@Cacheable(value = "availableMajors",
           unless = "#result.isEmpty()")
public List<String> getAvailableMajors() {
    try {
        // データベースから重複なしで専攻を取得
        List<String> majors = studentRepository
            .findDistinctMajors();

        // アルファベット順にソート
        return majors.stream()
            .filter(Objects::nonNull)
            .sorted()
            .collect(Collectors.toList());

    } catch (Exception e) {
        log.error("専攻一覧取得エラー", e);
        return Collections.emptyList();
    }
}

getAvailableStatuses()

List<String> getAvailableStatuses()

学生ステータス一覧の取得・フィルタリング機能付き

/**
 * 利用可能なステータス一覧を取得
 */
@Cacheable(value = "availableStatuses")
public List<String> getAvailableStatuses() {
    try {
        // 予定義されたステータス一覧
        List<String> predefinedStatuses = Arrays.asList(
            "ACTIVE", "INACTIVE", "GRADUATED",
            "SUSPENDED", "TRANSFERRED");

        // データベースから実際に使用されているステータスを取得
        List<String> actualStatuses = studentRepository
            .findDistinctStatuses();

        // 両方に存在するもののみ返す
        return predefinedStatuses.stream()
            .filter(actualStatuses::contains)
            .collect(Collectors.toList());

    } catch (Exception e) {
        log.error("ステータス一覧取得エラー", e);
        return Collections.emptyList();
    }
}

Ajax プレビュー用の検索件数取得メソッド

/**
 * Ajax プレビュー用：検索件数を取得
 */
public long countAdvancedSearch(
    StudentSearchCriteria criteria) {

    try {
        // 高度検索Specificationの作成
        Specification<Student> spec =
            StudentSpecifications.advancedSearch(criteria);

        // 件数のみ取得（高速）
        return studentRepository.count(spec);

    } catch (Exception e) {
        log.warn("検索件数取得エラー: {}", e.getMessage());
        return 0L;
    }
}

// リポジトリレベルでの最適化
@Repository
public interface StudentRepository extends JpaRepository<Student, Long>,
        JpaSpecificationExecutor<Student> {

    /**
     * 専攻一覧取得（インデックス活用）
     */
    @Query("SELECT DISTINCT s.major FROM Student s " +
           "WHERE s.major IS NOT NULL ORDER BY s.major")
    List<String> findDistinctMajors();

    /**
     * ステータス一覧取得（インデックス活用）
     */
    @Query("SELECT DISTINCT s.status FROM Student s " +
           "WHERE s.status IS NOT NULL")
    List<String> findDistinctStatuses();

    /**
     * 件数取得最適化（COUNT クエリ）
     */
    @Query("SELECT COUNT(s) FROM Student s WHERE " +
           "(:name IS NULL OR LOWER(s.name) LIKE LOWER(CONCAT('%', :name, '%')))")
    long countByNameContaining(@Param("name") String name);
}

@Configuration
@EnableCaching
public class CacheConfig {

    @Bean
    public CacheManager cacheManager() {
        ConcurrentMapCacheManager cacheManager =
            new ConcurrentMapCacheManager();

        // キャッシュ名の設定
        cacheManager.setCacheNames(Arrays.asList(
            "availableMajors",
            "availableStatuses",
            "searchResults"
        ));

        return cacheManager;
    }
}
