@PostMapping("/students") // Week 9.5
public String createStudent(
    @Valid @ModelAttribute StudentData studentData, // Week 9.1 + 9.3
    BindingResult bindingResult) { // Week 9.4
    
    // Week 9.4: エラーハンドリング
    if (bindingResult.hasErrors()) {
        return "students/form"; // Week 9.2画面に戻る
    }
    
    // Week 9.1: DTO→Entity変換
    Student student = studentData.toEntity();
    
    // Week 9.5: データ保存
    studentRepository.save(student);
    
    // Week 9.5: PRGパターン
    return "redirect:/students";
}
