@Test
public void testBulkLogicalDelete() {
    // テストデータ準備
    List<Long> studentIds = Arrays.asList(
        1L, 2L, 3L, 4L, 5L);

    // 一括削除実行
    BulkDeleteResult result =
        bulkDeleteService.bulkLogicalDelete(
            studentIds);

    // 結果検証
    assertThat(result.isAllSuccess()).isTrue();
    assertThat(result.getSuccessCount()).isEqualTo(5);
    assertThat(result.getFailureCount()).isEqualTo(0);

    // データベース状態確認
    for (Long id : studentIds) {
        Student student = studentRepository
            .findById(id).orElse(null);
        assertThat(student.getDeletedAt())
            .isNotNull();
    }

    System.out.println("✅ 一括論理削除テスト成功");
}

Step 3 テスト結果

  • ✓ @Transactional正常動作
  • ✓ 全件削除成功

// Step 4 API統合テスト
@Test
public void testBulkDeleteAPI() throws Exception {
    // リクエストパラメータ準備
    String studentIds = "1,2,3,4,5";

    // APIリクエスト実行
    mockMvc.perform(post("/api/students/bulk-delete")
        .param("studentIds", studentIds))
        .andExpect(status().isOk())
        .andExpect(jsonPath("$.success").value(true))
        .andExpect(jsonPath("$.statistics.total")
            .value(5))
        .andExpect(jsonPath("$.statistics.success")
            .value(5))
        .andExpected(jsonPath("$.message")
            .value("5件の学生を削除しました"));

    System.out.println("✅ API統合テスト成功");
}

// Step 5 Ajax統合テスト
async function testAjaxIntegration() {
    console.log('=== Ajax統合テスト開始 ===');

    // 1. 学生5件選択
    const testIds = [1, 2, 3, 4, 5];
    testIds.forEach(id => {
        selectionManager.updateSelectionState(
            id, true);
    });
    
    console.log('選択数:',
        selectionManager.getSelectedIds().length);

    // 2. 一括削除実行
    try {
        const response = await fetch(
            '/api/students/bulk-delete', {
            method: 'POST',
            headers: {
                'Content-Type':
                    'application/x-www-form-urlencoded'
            },
            body: `studentIds=${testIds.join(',')}`
        });

        const data = await response.json();
        
        // 3. レスポンス確認
        console.log('成功:', data.success);
        console.log('削除数:', data.statistics.success);
        console.log('メッセージ:', data.message);

        // 4. UI更新確認
        refreshStudentList();
        selectionManager.clearAll();
        
        console.log('✅ Ajax統合テスト成功');

    } catch (error) {
        console.error('❌ Ajax統合テスト失敗:', error);
    }
}
