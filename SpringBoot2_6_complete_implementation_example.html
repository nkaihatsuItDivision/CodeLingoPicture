📝 編集画面表示メソッド（GET）

showEditForm() メソッドの完全実装
/**
 * 編集画面表示処理
 * @param id 編集対象のStudent ID
 * @param model ビューに渡すデータモデル
 * @return 編集画面のテンプレート名
 */
@GetMapping("/students/{id}/edit")
public String showEditForm(@PathVariable Long id, Model model) {
    
    try {
        // 1. 指定されたIDのStudentエンティティを取得
        Student student = studentRepository.findById(id)
            .orElseThrow(() -> new IllegalArgumentException(
                "Invalid student Id:" + id));
        
        // 2. Week 9.1のfromEntity()メソッドでDTO変換
        StudentData studentData = StudentData.fromEntity(student);
        
        // 3. モデルに設定してビューに渡す
        model.addAttribute("studentData", studentData);
        
        return "students/edit";
        
    } catch (IllegalArgumentException e) {
        // 存在しないIDが指定された場合
        model.addAttribute("errorMessage",
            "指定されたデータが見つかりません。");
        return "redirect:/students";
    }
}

🔄 更新処理メソッド（PATCH）

updateStudent() メソッドの完全実装
/**
 * 学生情報更新処理
 * @param id 更新対象のStudent ID
 * @param studentData フォームから送信されたデータ
 * @param bindingResult バリデーション結果
 * @param model ビューに渡すデータモデル
 * @return 成功時は一覧画面、エラー時は編集画面
 */
@PatchMapping("/students/{id}")
public String updateStudent(@PathVariable Long id,
                           @Valid @ModelAttribute StudentData studentData,
                           BindingResult bindingResult,
                           Model model) {
    
    // 1. Week 9.3バリデーションエラーチェック
    if (bindingResult.hasErrors()) {
        model.addAttribute("errorMessage",
            "入力内容に不備があります。");
        return "students/edit";
    }
    
    try {
        // 2. DTOからEntityに変換
        Student student = studentData.toEntity();
        student.setId(id); // URLのIDを確実に設定
        
        // 3. 楽観ロック制御付きでデータベース更新
        studentRepository.save(student);
        
        // 4. 成功時は一覧画面にリダイレクト（PRGパターン）
        return "redirect:/students";
        
    } catch (OptimisticLockingFailureException e) {
        // 5. 楽観ロック競合例外処理
        return handleOptimisticLockingFailure(id, model);
        
    } catch (Exception e) {
        // 6. その他の予期しない例外
        model.addAttribute("errorMessage",
            "システムエラーが発生しました。");
        return "students/edit";
    }
}

/**
 * 楽観ロック例外処理の専用メソッド
 */
private String handleOptimisticLockingFailure(Long id, Model model) {
    model.addAttribute("errorMessage",
        "データが他のユーザーにより更新されています。
         最新データを確認してから再度実行してください。");
    
    try {
        // 最新データを再取得
        Student latestStudent = studentRepository.findById(id)
            .orElseThrow(() -> new IllegalArgumentException(
                "Invalid student Id:" + id));
        
        StudentData latestData = StudentData.fromEntity(latestStudent);
        model.addAttribute("studentData", latestData);
        
    } catch (IllegalArgumentException ex) {
        // レコードが削除されていた場合
        return "redirect:/students";
    }
    
    return "students/edit";
}
