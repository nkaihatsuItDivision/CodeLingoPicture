JavaScript - リアルタイムバリデーション
@PostMapping("/search")
public String executeSearch(@Valid @ModelAttribute("searchForm") StudentSearchForm searchForm,
                            BindingResult bindingResult,
                            RedirectAttributes redirectAttributes,
                            Model model) {

    // バリデーションエラーチェック
    if (bindingResult.hasErrors()) {
        // エラー情報をモデルに追加
        model.addAttribute("searchForm", searchForm);
        model.addAttribute("validationErrors", bindingResult.getAllErrors());
        return "students/search"; // 検索画面に戻る
    }

    // バリデーション成功時の処理
    StudentSearchCriteria criteria = searchForm.toCriteria();
    List<StudentDto> results = studentSearchService.searchStudents(criteria);

    // PRGパターンでリダイレクト
    redirectAttributes.addFlashAttribute("searchResults", results);
    return "redirect:/students/search/results";
}

document.addEventListener('DOMContentLoaded', function() {

<input type="text" th:field="*{name}"
       class="form-control"
       th:classappend="${#fields.hasErrors('name')} ? 'is-invalid' : ''"
       th:placeholder="${nameSearchType == 'partial'} ? '部分一致で検索' : '完全一致で検索'" />

<div th:if="${#fields.hasErrors('name')}" class="invalid-feedback">
    <span th:errors="*{name}"></span>
</div>

<!-- 成功フィードバック -->
<div th:unless="${#fields.hasErrors('name')}"
     th:if="${not #strings.isEmpty(searchForm.name)}"
     class="valid-feedback">
    入力形式が正しいです
</div>
