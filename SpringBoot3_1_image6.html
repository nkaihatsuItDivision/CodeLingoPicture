@Service
@Transactional
public class StudentServiceImpl implements StudentService {

@Autowired
private StudentRepository studentRepository;

public StudentDto createStudent(StudentForm form) throws ServiceException {
try {
Student student = convertToEntity(form);
Student saved = studentRepository.save(student);
return convertToDto(saved);
} catch (Exception e) {
throw new ServiceException("学生の登録に失敗しました", e);
}
}

public List<StudentDto> getAllStudents() {
return studentRepository.findAll().stream()
.map(this::convertToDto)
.collect(Collectors.toList());
}
}

🚀 統合検索サービス完全実装
@Service
@Transactional(readOnly = true)
public class StudentSearchServiceImpl implements StudentSearchService {

@Autowired
private StudentRepository studentRepository;

@Autowired
private StudentConverter studentConverter; // Week 9 変換ロジック再利用

/**
* 基本検索（Week 9 パターン継承）
*/
@Override
public List<StudentDto> searchStudents(StudentSearchCriteria criteria) throws ServiceException {
validateSearchCriteria(criteria); // Week 9 バリデーション

try {
Specification<Student> spec = StudentSpecifications.searchByCriteria(criteria);
List<Student> students = studentRepository.findAll(spec);

return students.stream()
.map(studentConverter::convertToDto) // Week 9 変換ロジック
.collect(Collectors.toList());

} catch (Exception e) {
log.error("検索エラー: criteria={}", criteria, e);
throw new ServiceException("学生検索に失敗しました", e);
}
}

/**
* 高度検索（Week 10.1 拡張機能）
*/
@Override
public List<StudentDto> searchStudentsAdvanced(StudentSearchCriteria criteria) throws ServiceException {
validateAdvancedCriteria(criteria);

try {
// カスタムリポジトリ活用
List<Student> students = studentRepository.findStudentsWithComplexCriteria(criteria);

return students.stream()
.map(studentConverter::convertToDto)
.collect(Collectors.toList());

} catch (Exception e) {
throw new ServiceException("高度検索に失敗しました", e);
}
}

/**
* Week 9 バリデーションパターン継承
*/
private void validateSearchCriteria(StudentSearchCriteria criteria) throws ServiceException {
if (criteria == null) {
throw new ServiceException("検索条件が指定されていません");
}

// 基本的な条件チェック
if (StringUtils.hasText(criteria.getName()) && criteria.getName().length() < 2) {
throw new ServiceException("名前は2文字以上で入力してください");
}

if (criteria.getMinGrade() != null && criteria.getMaxGrade() != null) {
if (criteria.getMinGrade() > criteria.getMaxGrade()) {
throw new ServiceException("最小成績は最大成績以下で入力してください");
}
}
}
}

Week 9 基本Controller
@Controller
@RequestMapping("/students")
public class StudentController {

@Autowired
private StudentService studentService;

@PostMapping
public String createStudent(@Valid @ModelAttribute StudentForm form,
BindingResult result,
RedirectAttributes redirectAttributes) {
if (result.hasErrors()) {
return "student/form";
}

try {
studentService.createStudent(form);
redirectAttributes.addFlashAttribute("successMessage", "学生を登録しました");
return "redirect:/students";
} catch (ServiceException e) {
redirectAttributes.addFlashAttribute("errorMessage", e.getMessage());
return "redirect:/students/form";
}
}
}
Week 10.1 検索Controller拡張ENHANCED
@Controller
@RequestMapping("/students")
public class StudentController {

@Autowired
private StudentService studentService;

@Autowired
private StudentSearchService studentSearchService; // 検索サービス追加

// Week 9 PRGパターン継承
@PostMapping("/search")
public String searchStudents(@Valid @ModelAttribute StudentSearchCriteria criteria,
BindingResult result,
Model model,
RedirectAttributes redirectAttributes) {

// Week 9 バリデーション継承
if (result.hasErrors()) {
model.addAttribute("criteria", criteria);
return "student/search";
}

try {
// Week 10.1 検索機能
List<StudentDto> students = studentSearchService.searchStudents(criteria);

model.addAttribute("students", students);
model.addAttribute("criteria", criteria);
model.addAttribute("resultCount", students.size());

return "student/search-results";

} catch (ServiceException e) {
// Week 9 エラーハンドリング継承
model.addAttribute("errorMessage", e.getMessage());
model.addAttribute("criteria", criteria);
return "student/search";
}
}
}
