StudentSearchForm.java - Bean Validation設計
@Component
public class StudentSearchForm {
    @Size(max = 100, message = "学生名は100文字以内で入力してください")
    private String name;

    @Min(value = 1, message = "学年は1以上で入力してください")
    @Max(value = 6, message = "学年は6以下で入力してください")
    private Integer grade;

    @Email(message = "正しいメールアドレス形式で入力してください")
    private String email;

    @Pattern(regexp = "^(partial|exact)$", message = "検索タイプが不正です")
    private String nameSearchType = "partial";

    // Week 10.1のStudentSearchCriteriaへの変換
    public StudentSearchCriteria toCriteria() {
        return StudentSearchCriteria.builder()
            .name(this.name)
            .grade(this.grade)
            .email(this.email)
            .nameSearchType(this.nameSearchType)
            .build();
    }
}

Thymeleafバインディング設計

<form th:action="@{/students/search}" th:object="${searchForm}" method="post">
    <div class="row">
        <!-- 学生名入力 -->
        <div class="col-md-6">
            <label class="form-label">学生名</label>
            <input type="text" th:field="*{name}"
                   class="form-control"
                   th:classappend="${#fields.hasErrors('name')} ? 'is-invalid' : ''"
                   placeholder="学生名を入力" />
            <div th:if="${#fields.hasErrors('name')}" class="invalid-feedback">
                <span th:errors="*{name}"></span>
            </div>
        </div>

        <!-- 検索タイプ選択 -->
        <div class="col-md-6">
            <label class="form-label">検索タイプ</label>
            <div class="form-check-inline">
                <input type="radio" th:field="*{nameSearchType}"
                       value="partial" class="form-check-input" />
                <label class="form-check-label">部分一致</label>
            </div>
        </div>
    </div>
</form>
