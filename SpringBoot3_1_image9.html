// 動的検索実装例
public static Specification<Student> searchByCriteria(StudentSearchCriteria criteria) {
return Specification.where(nameContains(criteria.getName()))
.and(majorEquals(criteria.getMajor()))
.and(gradeInRange(criteria.getMinGrade(), criteria.getMaxGrade()))
.and(statusEquals(criteria.getStatus()))
.and(enrollmentDateInRange(criteria.getStartDate(), criteria.getEndDate()));
}

private static Specification<Student> gradeInRange(Integer minGrade, Integer maxGrade) {
return (root, query, cb) -> {
if (minGrade == null && maxGrade == null) return null;
if (minGrade == null) return cb.lessThanOrEqualTo(root.get("grade"), maxGrade);
if (maxGrade == null) return cb.greaterThanOrEqualTo(root.get("grade"), minGrade);
return cb.between(root.get("grade"), minGrade, maxGrade);
};
}

@Service
@Transactional(readOnly = true)
public class StudentSearchServiceImpl implements StudentSearchService {

@Autowired
private StudentRepository studentRepository;

public List<StudentDto> searchStudents(StudentSearchCriteria criteria) throws ServiceException {
try {
validateSearchCriteria(criteria); // Week 9 バリデーション

Specification<Student> spec = StudentSpecifications.searchByCriteria(criteria);
List<Student> students = studentRepository.findAll(spec);

return students.stream()
.map(this::convertToDto) // Week 9 DTOパターン
.collect(Collectors.toList());

} catch (Exception e) {
throw new ServiceException("学生検索に失敗しました", e); // Week 9 エラーハンドリング
}
}
}
