// リアルタイム検索の実装
function executeRealTimeSearch() {
    const form = document.querySelector('#advancedSearchForm');
    const formData = new FormData(form);
    
    // URLSearchParams でクエリパラメータ構築
    const params = new URLSearchParams();
    for (let [key, value] of formData.entries()) {
        if (value.trim() !== '') {
            params.append(key, value);
        }
    }

    // Ajax による非同期通信
    fetch('/students/api/search/preview?' + params.toString(), {
        method: 'GET',
        headers: {
            'Content-Type': 'application/json',
            'X-Requested-With': 'XMLHttpRequest'
        }
    })
    .then(response => response.json())
    .then(data => {
        updateSearchPreview(data);
    })
    .catch(error => {
        console.error('検索エラー:', error);
        showErrorMessage('検索に失敗しました');
    });
}

入力頻度を制御してサーバー負荷を軽減する最適化処理

// デバウンス機能の実装
let searchTimeout;
const DEBOUNCE_DELAY = 300; // 300ms

function debounceSearch() {
    // 既存のタイマーをクリア
    clearTimeout(searchTimeout);

    // 新しいタイマーを設定
    searchTimeout = setTimeout(() => {
        executeRealTimeSearch();
    }, DEBOUNCE_DELAY);
}

// イベントリスナーの設定
document.addEventListener('DOMContentLoaded', function() {
    // 入力フィールドのイベント監視
    const searchInputs = document.querySelectorAll(
        'input[type="text"], input[type="number"], ' +
        'input[type="date"], select, input[type="checkbox"]');

    searchInputs.forEach(input => {
        input.addEventListener('input', debounceSearch);
        input.addEventListener('change', debounceSearch);
    });
});

// 検索プレビューの更新
function updateSearchPreview(data) {
    const resultCountElement = document.getElementById('result-count');
    const loadingElement = document.getElementById('loading');

    // ローディング非表示
    loadingElement.style.display = 'none';

    // 結果件数の更新
    if (data.count !== undefined) {
        const formattedCount = data.count.toLocaleString();
        resultCountElement.textContent =
            `検索結果: ${formattedCount}件`;
    
        // 件数に応じた色変更
        if (data.count === 0) {
            resultCountElement.style.color = '#dc3545';
        } else if (data.count > 1000) {
            resultCountElement.style.color = '#fd7e14';
        } else {
            resultCountElement.style.color = '#28a745';
        }
    }
}

// 高度なデバウンス実装
class SearchDebouncer {
    constructor(callback, delay = 300) {
        this.callback = callback;
        this.delay = delay;
        this.timeoutId = null;
        this.isSearching = false;
    }

    execute(...args) {
        // 既存のタイマーをクリア
        clearTimeout(this.timeoutId);

        // ローディング表示
        this.showLoading();

        // 新しいタイマー設定
        this.timeoutId = setTimeout(() => {
            this.isSearching = true;
            this.callback.apply(this, args)
                .finally(() => {
                    this.isSearching = false;
                    this.hideLoading();
                });
        }, this.delay);
    }

    showLoading() {
        document.getElementById('loading').style.display = 'inline-block';
    }

    hideLoading() {
        document.getElementById('loading').style.display = 'none';
    }
}

// 選択肢の動的更新機能
function updateDependentSelections() {
    const majorSelect = document.getElementById('major');
    const gradeSelect = document.getElementById('grade');
    const statusSelect = document.getElementById('status');

    // 専攻選択時に学年選択肢を更新
    majorSelect.addEventListener('change', function() {
        const selectedMajor = this.value;

        // Ajax で該当専攻の学年一覧を取得
        fetch(`/students/api/grades?major=${selectedMajor}`)
            .then(response => response.json())
            .then(grades => {
                // select オプションを動的更新
                gradeSelect.innerHTML =
                    '<option value="">すべて</option>';
                
                grades.forEach(grade => {
                    const option = document.createElement('option');
                    option.value = grade;
                    option.textContent = `${grade}年生`;
                    gradeSelect.appendChild(option);
                });

                // フェードイン効果
                gradeSelect.style.opacity = '0.5';
                setTimeout(() => {
                    gradeSelect.style.opacity = '1';
                }, 100);
            });
    });
}
