Specification生成
検索条件に基づいて動的にSpecificationオブジェクトを生成。null値の条件は無視して柔軟な検索を実現
public static Specification<Student> searchByCriteria(StudentSearchCriteria criteria) {
return (root, query, criteriaBuilder) -> {
List<Predicate> predicates = new ArrayList<>();

// 名前の部分一致検索
if (criteria.getName() != null) {
predicates.add(criteriaBuilder.like(
criteriaBuilder.lower(root.get("name")),
"%" + criteria.getName().toLowerCase() + "%"
));
}

// 専攻の完全一致検索
if (criteria.getMajor() != null) {
predicates.add(criteriaBuilder.equal(root.get("major"), criteria.getMajor()));
}

// 成績範囲検索
if (criteria.getMinGrade() != null) {
predicates.add(criteriaBuilder.greaterThanOrEqualTo(root.get("grade"), criteria.getMinGrade()));
}
if (criteria.getMaxGrade() != null) {
predicates.add(criteriaBuilder.lessThanOrEqualTo(root.get("grade"), criteria.getMaxGrade()));
}

return criteriaBuilder.and(predicates.toArray(new Predicate[0]));
};
}

// 条件の組み合わせパターン例
private static Specification<Student> buildComplexSpecification(StudentSearchCriteria criteria) {
return Specification.where(nameContains(criteria.getName()))
.and(majorEquals(criteria.getMajor()))
.and(gradeInRange(criteria.getMinGrade(), criteria.getMaxGrade()))
.and(enrollmentDateInRange(criteria.getStartDate(), criteria.getEndDate()))
.and(statusEquals(criteria.getStatus()));
}

// 個別Specification実装
private static Specification<Student> nameContains(String name) {
return (root, query, cb) -> name == null ? null :
cb.like(cb.lower(root.get("name")), "%" + name.toLowerCase() + "%");
}

private static Specification<Student> gradeInRange(Integer minGrade, Integer maxGrade) {
return (root, query, cb) -> {
if (minGrade == null && maxGrade == null) return null;
if (minGrade == null) return cb.lessThanOrEqualTo(root.get("grade"), maxGrade);
if (maxGrade == null) return cb.greaterThanOrEqualTo(root.get("grade"), minGrade);
return cb.between(root.get("grade"), minGrade, maxGrade);
};
}

// Service層での実行
@Service
@Transactional(readOnly = true)
public class StudentSearchServiceImpl implements StudentSearchService {

@Autowired
private StudentRepository studentRepository;

public List<StudentDto> searchStudents(StudentSearchCriteria criteria) {
Specification<Student> spec = StudentSpecifications.searchByCriteria(criteria);
List<Student> students = studentRepository.findAll(spec);

return students.stream()
.map(this::convertToDto)
.collect(Collectors.toList());
}
}
🔧 Specificationパターンの強力な機能
// OR条件の実装
public static Specification<Student> majorInList(List<String> majors) {
return (root, query, cb) -> {
if (majors == null || majors.isEmpty()) return null;
return root.get("major").in(majors);
};
}

// 複雑な条件組み合わせ
public static Specification<Student> complexSearch(StudentSearchCriteria criteria) {
return Specification.where(nameContains(criteria.getName()))
.and(majorInList(criteria.getMajors()))
.and(gradeInRange(criteria.getMinGrade(), criteria.getMaxGrade()))
.and(activeStudentsOnly())
.and(enrolledAfter(criteria.getEnrollmentDate()));
}

// JOIN検索の実装
public static Specification<Student> hasScoreAbove(Double minScore) {
return (root, query, cb) -> {
Join<Student, Score> scoreJoin = root.join("scores");
return cb.greaterThan(scoreJoin.get("value"), minScore);
};
}
