列ヘッダーに動的なソートリンクを生成し、現在のソート状態をアイコンで表示

Controllerでのソート処理
@GetMapping("/students")
public String list(
    @RequestParam(defaultValue = "0") int page,
    @RequestParam(defaultValue = "10") int size,
    @RequestParam(defaultValue = "id") String sort,
    @RequestParam(defaultValue = "asc") String direction,
    Model model) {

    Sort.Direction dir = Sort.Direction.fromString(direction);
    Sort sortObj = Sort.by(dir, sort);
    Pageable pageable = PageRequest.of(page, size, sortObj);

    Page<Student> students = studentService.findAll(pageable);
    model.addAttribute("page", students);

    return "students/list";
}

リクエストパラメータからSortオブジェクトを生成し、Pageableに適用

Repository自動対応
public interface StudentRepository extends JpaRepository<Student, Long> {

    // Spring Data JPAが自動でソート機能を提供
    Page<Student> findAll(Pageable pageable);

    // カスタム検索でもソート対応
    Page<Student> findByNameContaining(String name, Pageable pageable);

    // @Query使用時もソート対応
    @Query("SELECT s FROM Student s WHERE s.grade = :grade")
    Page<Student> findByGrade(@Param("grade") String grade, Pageable pageable);
}

Repositoryでは特別な実装不要。PageableでSort情報が自動的に処理される

// 複数ソート条件の構築
Sort sort = Sort.by(
    Sort.Order.desc("grade"),
    Sort.Order.asc("age"),
    Sort.Order.asc("name")
);

Pageable pageable = PageRequest.of(page, size, sort);
Page<Student> result = studentRepository.findAll(pageable);
