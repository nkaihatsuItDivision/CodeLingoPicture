💻 Week 10.1-10.5 統合コード例
@RestController
@RequestMapping("/api/students")
public class StudentIntegratedController {

    @Autowired
    private StudentSearchService searchService; // Week 10.1-10.4
    @Autowired
    private StudentDeleteService deleteService; // Week 10.5

    // 高度検索機能（Week 10.1-10.4統合）
    @PostMapping("/search")
    public Page<StudentDto> advancedSearch(
        @RequestBody StudentSearchCriteria criteria,
        Pageable pageable) {
        return searchService.searchWithComplexCriteria(criteria, pageable);
    }

    // 検索連携削除機能（Week 10.5統合）
    @DeleteMapping("/{id}")
    @Transactional
    public ResponseEntity<DeleteResult> deleteStudent(
        @PathVariable Long id,
        @RequestParam String deleteType,
        @RequestParam String deleteReason,
        Principal principal) {

        // 1. 検索機能で削除対象確認
        StudentDto student = searchService.findById(id);
        if (student == null) {
            return ResponseEntity.notFound().build();
        }

        // 2. 削除権限確認
        if (!hasDeletePermission(principal, id)) {
            return ResponseEntity.status(403).build();
        }

        // 3. 削除実行（論理削除・物理削除対応）
        DeleteResult result = deleteService.executeDelete(
            id, deleteType, deleteReason, principal.getName());

        return ResponseEntity.ok(result);
    }
}
