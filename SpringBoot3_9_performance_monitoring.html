🎯 PerformanceMonitoringAspect の実装
⚡
@Around アスペクト
@Aspect
@Component
public class PerformanceMonitoringAspect {

    private final MeterRegistry meterRegistry;

    @Around("@annotation(PerformanceMonitoring)")
    public Object monitor(ProceedingJoinPoint joinPoint)
            throws Throwable {

        Timer.Sample sample = Timer.start(meterRegistry);

        try {
            Object result = joinPoint.proceed();
            recordSuccess(joinPoint);
            return result;
        } catch (Exception e) {
            recordError(joinPoint, e);
            throw e;
        } finally {
            sample.stop(Timer.builder("method.execution")
                .register(meterRegistry));
        }
    }
}

メソッド実行を自動的に監視し、実行時間と成功/失敗を記録

📊
Timer.Sample
@Service
public class CustomMetricsService {

    private final MeterRegistry meterRegistry;
    private final Counter requestCounter;
    private final Timer responseTimer;

    public CustomMetricsService(MeterRegistry registry) {
        this.meterRegistry = registry;
        this.requestCounter = Counter.builder("api.requests")
            .description("API request count")
            .register(registry);
        this.responseTimer = Timer.builder("api.response.time")
            .description("API response time")
            .register(registry);
    }

    public void recordApiCall(String endpoint) {
        requestCounter.increment(Tags.of("endpoint", endpoint));
    }
}
