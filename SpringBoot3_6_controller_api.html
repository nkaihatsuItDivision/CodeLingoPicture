@RestController
@RequestMapping("/api/students")
public class StudentBulkDeleteController {

    @Autowired
    private StudentBulkDeleteService bulkDeleteService;

    /**
     * 一括削除API
     * POST /api/students/bulk-delete
     */
    @PostMapping("/bulk-delete")
    public ResponseEntity<Map<String, Object>>
            bulkDelete(
        @RequestParam("studentIds")
            List<Long> studentIds,
        @RequestParam(value = "deleteType",
            defaultValue = "logical")
            String deleteType) {

        // サービス層呼び出し
        BulkDeleteResult result =
            bulkDeleteService.bulkLogicalDelete(
                studentIds);

        // JSONレスポンス構築
        return createJsonResponse(result);
    }
}

// リクエスト処理とバリデーション
private ResponseEntity<Map<String, Object>>
        processRequest(
            List<Long> studentIds,
            String deleteType) {

    try {
        // 1. 入力バリデーション
        if (studentIds == null || studentIds.isEmpty()) {
            return createErrorResponse(
                "削除対象が選択されていません",
                HttpStatus.BAD_REQUEST);
        }

        // 2. サイズ制限チェック
        if (studentIds.size() > 100) {
            return createErrorResponse(
                "一度に削除できるのは100件までです",
                HttpStatus.BAD_REQUEST);
        }

        // 3. ID重複除去
        List<Long> uniqueIds = studentIds.stream()
            .distinct()
            .collect(Collectors.toList());

        // 4. サービス実行
        BulkDeleteResult result =
            executeDeleteOperation(
                uniqueIds, deleteType);

        return createSuccessResponse(result);

    } catch (Exception e) {
        return createErrorResponse(
            e.getMessage(),
            HttpStatus.INTERNAL_SERVER_ERROR);
    }
}

リクエスト処理技術

// JSONレスポンス構築
private ResponseEntity<Map<String, Object>>
        createJsonResponse(BulkDeleteResult result) {

    Map<String, Object> response = new HashMap<>();

    // 1. 基本情報
    response.put("success", result.isAllSuccess());
    response.put("timestamp",
        LocalDateTime.now().toString());

    // 2. 統計情報
    Map<String, Integer> statistics = new HashMap<>();
    statistics.put("total", result.getTotalCount());
    statistics.put("success", result.getSuccessCount());
    statistics.put("failure", result.getFailureCount());
    response.put("statistics", statistics);

    // 3. 成功したID一覧
    response.put("successIds", result.getSuccessIds());

    // 4. エラー詳細（失敗がある場合）
    if (!result.isAllSuccess()) {
        response.put("errors", result.getFailures());
    }

    // 5. メッセージ生成
    response.put("message", createMessage(result));

    // 6. HTTPステータス決定
    HttpStatus status = result.isAllSuccess() ?
        HttpStatus.OK : HttpStatus.PARTIAL_CONTENT;

    return new ResponseEntity<>(response, status);
}

  • 構造化JSON設計

  • エラー情報の詳細化

Ajax通信フロー
フロントエンドとの非同期通信による一括削除機能の完全統合
// Ajax通信実装（JavaScript側）
function executeBulkDelete() {
    const selectedIds = selectionManager.getSelectedIds();

    if (selectedIds.length === 0) {
        alert('削除する学生を選択してください');
        return;
    }

    // 確認ダイアログ
    if (!confirm(
        `選択した${selectedIds.length}件を削除しますか？`)) {
        return;
    }

    // Ajax送信
    fetch('/api/students/bulk-delete', {
        method: 'POST',
        headers: {
            'Content-Type':
                'application/x-www-form-urlencoded'
        },
        body: `studentIds=${selectedIds.join(',')}`
    })
    .then(response => response.json())
    .then(data => handleDeleteResponse(data))
    .catch(error => handleDeleteError(error));
}

// レスポンス処理
function handleDeleteResponse(data) {
    if (data.success) {
        showSuccessMessage(data.message);
        refreshStudentList();
        selectionManager.clearAll();
    } else {
        showPartialErrorMessage(data);
    }
}
