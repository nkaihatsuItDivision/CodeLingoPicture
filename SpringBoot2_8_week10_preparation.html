@Service
@Transactional
public class StudentService {

    @Autowired
    private StudentRepository repository;

    // ✅ 基本CRUD操作完成
    public void createStudent(StudentData data) {
        validateBusinessRules(data);
        Student student = convertToEntity(data);
        repository.save(student);
    }

    public void updateStudent(Long id, StudentData data) {
        Student student = getStudentById(id);
        updateEntity(student, data);
        repository.save(student);
    }

    // 🔧 Week 10での拡張基盤
    private Student getStudentById(Long id) {
        return repository.findById(id)
            .orElseThrow(() ->
                new DataNotFoundException("学生", id));
    }
}

@Service

public class StudentService {

        // 基盤活用：既存のService・DI・例外処理
        return repository.findByQuery(query);
    }

    // 🗑️ Week 10.4-10.5: 安全な削除機能
    public void deleteStudent(Long id) {
        // 基盤活用：例外処理・トランザクション
        Student student = getStudentById(id);
        validateDeletion(student);
        repository.delete(student);
    }

    // 📊 Week 10.6-10.7: RESTful API
    public Page<Student> findStudentsWithPaging(
            StudentQuery query, Pageable pageable) {
        // 基盤活用：既存アーキテクチャの拡張
        return repository.findByQueryWithPaging(query, pageable);
    }
}
