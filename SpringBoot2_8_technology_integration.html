📋 Service層でのビジネスロジック統合
@Service
@Transactional
public class StudentService {

    @Autowired
    private StudentRepository repository;

    public void createStudent(StudentData data) {
        // Week 9.1-9.4: DTO・Validation
        validateBusinessRules(data);
        
        // Week 9.5-9.6: データ変換・保存
        Student student = convertToEntity(data);
        repository.save(student);
    }

    // Week 9.8: 例外処理統一化
    private void validateBusinessRules(StudentData data) {
        if (data.getAge() < 18) {
            throw new ValidationException("18歳以上");
        }
    }
}
🎮 Controller層での統合制御
@Controller
public class StudentController {

    @Autowired
    private StudentService studentService;

    @PostMapping("/students")
    public String create(@Valid StudentData data,
                       BindingResult result,
                       RedirectAttributes ra) {
        // Week 9.3-9.4: バリデーション
        if (result.hasErrors()) {
            return "student-form";
        }

        try {
            // Week 9.8: Service層委譲
            studentService.createStudent(data);
            // Week 9.7: 成功メッセージ
            ra.addFlashAttribute("success", "登録完了");
        } catch (ServiceException e) {
            // Week 9.8: 例外処理統一化
            ra.addFlashAttribute("error", e.getMessage());
        }
        // Week 9.7: PRGパターン
        return "redirect:/students";
    }
}
