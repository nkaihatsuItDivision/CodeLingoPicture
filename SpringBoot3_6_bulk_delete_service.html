@Transactional一括処理
複数データを安全に一括処理するトランザクション制御の実装
@Service
@Transactional
public class StudentBulkDeleteServiceImpl
        implements StudentBulkDeleteService {

    @Autowired
    private StudentDeleteService deleteService;

    @Override
    @Transactional(rollbackFor = Exception.class)
    public BulkDeleteResult bulkLogicalDelete(
            List<Long> studentIds) {

        BulkDeleteResult result =
            new BulkDeleteResult();
        
        for (Long studentId : studentIds) {
            try {
                // Week 10.5の削除サービス活用
                deleteService.logicalDelete(studentId);
                result.addSuccess(studentId);
            } catch (Exception e) {
                result.addFailure(studentId,
                    e.getMessage());
            }
        }
        
        return result;
    }
}

// BulkDeleteResult.java
@Data
public class BulkDeleteResult {
    private List<Long> successIds = new ArrayList<>();
    private List<BulkDeleteError> failures =
        new ArrayList<>();
    private int totalCount;
    private int successCount;
    private int failureCount;

    public void addSuccess(Long studentId) {
        successIds.add(studentId);
        successCount++;
    }

    public void addFailure(Long studentId,
                        String errorMessage) {
        failures.add(new BulkDeleteError(
            studentId, errorMessage));
        failureCount++;
    }

    public boolean isAllSuccess() {
        return failureCount == 0;
    }

    public boolean hasPartialFailure() {
        return successCount > 0 && failureCount > 0;
    }
}

@Override
@Transactional(rollbackFor = Exception.class)
public BulkDeleteResult optimizedBulkDelete(
        List<Long> studentIds) {

    // 1. 事前バリデーション（一括）
    List<Long> validIds =
        validateStudentIds(studentIds);

    // 2. バッチサイズでの分割処理
    final int BATCH_SIZE = 100;
    BulkDeleteResult result = new BulkDeleteResult();

    for (int i = 0; i < validIds.size();
         i += BATCH_SIZE) {
        
        int endIndex = Math.min(
            i + BATCH_SIZE, validIds.size());
        
        List<Long> batch = validIds.subList(i, endIndex);
        
        // 3. バッチごとの削除実行
        BulkDeleteResult batchResult =
            processBatchDelete(batch);
        
        // 4. 結果の統合
        result.merge(batchResult);
    }

    return result;
}
