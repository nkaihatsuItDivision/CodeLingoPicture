データベースインデックス設計
1. 基本インデックス設定
検索頻度の高いカラムに適切なインデックスを設定し、クエリ実行速度を劇的に向上させます。
@Entity
@Table(name = "students", indexes = {
@Index(name = "idx_student_name", columnList = "name"),
@Index(name = "idx_student_email", columnList = "email"),
@Index(name = "idx_student_major", columnList = "major"),
@Index(name = "idx_student_grade", columnList = "grade"),
@Index(name = "idx_student_status", columnList = "status"),
@Index(name = "idx_student_enrollment_date", columnList = "enrollment_date")
})
public class Student {
@Id
@GeneratedValue(strategy = GenerationType.IDENTITY)
private Long id;

@Column(name = "name", nullable = false)
private String name;

@Column(name = "email", unique = true)
private String email;

// その他フィールド
}
2. 複合インデックス戦略
複数カラムの組み合わせ検索に対応する複合インデックスで、より高度な最適化を実現。
@Table(name = "students", indexes = {
// 複合インデックス：検索頻度と選択性を考慮した順序
@Index(name = "idx_major_grade", columnList = "major, grade"),
@Index(name = "idx_status_major_grade", columnList = "status, major, grade"),
@Index(name = "idx_enrollment_date_major", columnList = "enrollment_date, major"),

// 範囲検索対応
@Index(name = "idx_grade_enrollment_date", columnList = "grade, enrollment_date"),

// 部分一致検索最適化（主要検索条件のみ）
@Index(name = "idx_name_major", columnList = "name, major")
})
⚠️ インデックス設計の注意点
インデックスは検索性能を向上させますが、INSERT/UPDATE/DELETE性能を低下させる可能性があります。実際の使用パターンを分析して適切なバランスを取ることが重要です。

@Repository
public interface StudentRepository extends JpaRepository<Student, Long> {

@EntityGraph("Student.withScores")
List<Student> findByMajor(String major);

@EntityGraph("Student.withAllRelations")

2. JOIN FETCHによる効率的取得
JPQLのJOIN FETCH句を使用して、必要な関連データを一括取得。
@Repository
public interface StudentRepository extends JpaRepository<Student, Long> {

// 成績データを含む学生検索
@Query("SELECT DISTINCT s FROM Student s " +
"LEFT JOIN FETCH s.scores " +
"WHERE s.major = :major")
List<Student> findByMajorWithScores(@Param("major") String major);

// 複数関連エンティティの一括取得
@Query("SELECT DISTINCT s FROM Student s " +
"LEFT JOIN FETCH s.scores sc " +
"LEFT JOIN FETCH s.enrollmentHistory eh " +
"WHERE s.status = 'ACTIVE' AND s.grade >= :minGrade")
List<Student> findActiveStudentsWithAllData(@Param("minGrade") Integer minGrade);

// 条件付き関連データ取得
@Query("SELECT s FROM Student s " +
"LEFT JOIN FETCH s.scores sc " +
"WHERE s.name LIKE %:name% AND sc.value >= :minScore")
List<Student> findByNameWithHighScores(@Param("name") String name,
@Param("minScore") Double minScore);
}

効率的なクエリ設計原則
1. 選択性を考慮した条件順序
WHERE句の条件順序を最適化し、データベースエンジンの効率を最大化します。
// 最適化されたSpecification実装
public static Specification<Student> optimizedSearch(StudentSearchCriteria criteria) {
return (root, query, cb) -> {
List<Predicate> predicates = new ArrayList<>();

// 1. 最も選択性の高い条件から順番に追加
// メールアドレス（ユニーク制約）
if (StringUtils.hasText(criteria.getEmail())) {
predicates.add(cb.equal(root.get("email"), criteria.getEmail()));
}

// 2. ステータス（インデックス有り・高選択性）
if (StringUtils.hasText(criteria.getStatus())) {
predicates.add(cb.equal(root.get("status"), criteria.getStatus()));
}

// 3. 専攻（インデックス有り・中選択性）
if (StringUtils.hasText(criteria.getMajor())) {
predicates.add(cb.equal(root.get("major"), criteria.getMajor()));
}

// 4. 範囲検索（インデックス活用）
if (criteria.getMinGrade() != null || criteria.getMaxGrade() != null) {
if (criteria.getMinGrade() != null && criteria.getMaxGrade() != null) {
predicates.add(cb.between(root.get("grade"), criteria.getMinGrade(), criteria.getMaxGrade()));
} else if (criteria.getMinGrade() != null) {
predicates.add(cb.greaterThanOrEqualTo(root.get("grade"), criteria.getMinGrade()));
} else {
predicates.add(cb.lessThanOrEqualTo(root.get("grade"), criteria.getMaxGrade()));
}
}

// 5. 部分一致検索（最も負荷が高い）
if (StringUtils.hasText(criteria.getName())) {
predicates.add(cb.like(cb.lower(root.get("name")),
"%" + criteria.getName().toLowerCase() + "%"));
}

return cb.and(predicates.toArray(new Predicate[0]));
};
}
2. バッチサイズ最適化

@Entity
@BatchSize(size = 25) // 関連エンティティの一括取得サイズ

}
