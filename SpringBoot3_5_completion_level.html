@PreAuthorize("hasRole('ADMIN') or hasRole('MANAGER')")
@PostMapping("/delete")
public String deleteStudent(
    @RequestParam Long studentId,
    @RequestParam String deleteType,
    @RequestParam String deleteReason,
    Principal principal) {
    
    // 削除権限チェック
    validateDeletePermission(principal, studentId);
    
    // 削除実行
    deleteService.delete(studentId, deleteType,
                        deleteReason, principal.getName());
    return "redirect:/students";
}
