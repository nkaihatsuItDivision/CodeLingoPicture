package com.example.repository;

import java.util.List;
import com.example.dto.StudentSearchCriteria;
import com.example.entity.Student;

/**
* StudentRepository用カスタムインターフェース
* 複雑な検索ロジック・統計処理を定義
*/
public interface StudentRepositoryCustom {

/**
* 複雑な条件による学生検索
*/
List<Student> findStudentsWithComplexCriteria(StudentSearchCriteria criteria);

*/

/**
* 上位N位の学生を取得
*/
List<Student> findTopStudentsByScore(int limit);

* 月別入学者数統計
*/
List<Object[]> getMonthlyEnrollmentStatistics(int year);
}
2
実装クラス作成
EntityManagerを活用してカスタムインターフェースの実装クラスを作成。命名規則「インターフェース名 + Impl」を遵守
package com.example.repository;

import org.springframework.stereotype.Repository;
import javax.persistence.EntityManager;
import javax.persistence.PersistenceContext;
import javax.persistence.TypedQuery;
import javax.persistence.criteria.*;
import java.util.ArrayList;
import java.util.List;

@Repository
public class StudentRepositoryCustomImpl implements StudentRepositoryCustom {

@PersistenceContext
private EntityManager entityManager;

@Override
public List<Student> findStudentsWithComplexCriteria(StudentSearchCriteria criteria) {
CriteriaBuilder cb = entityManager.getCriteriaBuilder();
CriteriaQuery<Student> query = cb.createQuery(Student.class);
Root<Student> root = query.from(Student.class);

List<Predicate> predicates = new ArrayList<>();

// 複雑な条件構築
if (criteria.getName() != null) {
predicates.add(cb.like(cb.lower(root.get("name")),
"%" + criteria.getName().toLowerCase() + "%"));
}

if (criteria.getMajors() != null && !criteria.getMajors().isEmpty()) {
predicates.add(root.get("major").in(criteria.getMajors()));
}

// スコアランキング条件
if (criteria.getMinScore() != null) {
Subquery<Double> subquery = query.subquery(Double.class);
Root<Score> scoreRoot = subquery.from(Score.class);
subquery.select(cb.avg(scoreRoot.get("value")))
.where(cb.equal(scoreRoot.get("student"), root));
predicates.add(cb.greaterThan(subquery, criteria.getMinScore()));
}

query.where(cb.and(predicates.toArray(new Predicate[0])));
query.orderBy(cb.desc(root.get("grade")), cb.asc(root.get("name")));

TypedQuery<Student> typedQuery = entityManager.createQuery(query);
return typedQuery.getResultList();
}
}

統計・集計クエリ実装
Criteria APIを活用した高度な統計処理・集計クエリの実装
@Override
public List<Object[]> getGradeStatisticsByMajor() {
CriteriaBuilder cb = entityManager.getCriteriaBuilder();
CriteriaQuery<Object[]> query = cb.createQuery(Object[].class);
Root<Student> root = query.from(Student.class);

query.multiselect(
root.get("major"), // 専攻
cb.count(root), // 学生数
cb.avg(root.get("grade")), // 平均成績
cb.max(root.get("grade")), // 最高成績
cb.min(root.get("grade")) // 最低成績
);

query.groupBy(root.get("major"));
query.orderBy(cb.desc(cb.avg(root.get("grade"))));

return entityManager.createQuery(query).getResultList();
}

@Override
public List<Student> findTopStudentsByScore(int limit) {
CriteriaBuilder cb = entityManager.getCriteriaBuilder();
CriteriaQuery<Student> query = cb.createQuery(Student.class);
Root<Student> root = query.from(Student.class);
Join<Student, Score> scoreJoin = root.join("scores");

query.select(root)
.groupBy(root.get("id"))
.orderBy(cb.desc(cb.avg(scoreJoin.get("value"))));

return entityManager.createQuery(query)
.setMaxResults(limit)
.getResultList();
}

JpaRepositoryとの統合
カスタムインターフェースをJpaRepositoryと統合し、すべての機能を一つのリポジトリで提供
package com.example.repository;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.JpaSpecificationExecutor;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import org.springframework.stereotype.Repository;
import com.example.entity.Student;
import java.util.List;

/**
* 統合Studentリポジトリインターフェース
* JpaRepository + JpaSpecificationExecutor + カスタム機能
*/
@Repository
public interface StudentRepository extends
JpaRepository<Student, Long>,
JpaSpecificationExecutor<Student>,
StudentRepositoryCustom { // カスタムインターフェース継承

// 基本的なクエリメソッド
List<Student> findByMajor(String major);

List<Student> findByGradeBetween(Integer minGrade, Integer maxGrade);

// @Queryアノテーション活用
@Query("SELECT s FROM Student s WHERE s.name LIKE %:name% ORDER BY s.grade DESC")
List<Student> findByNameContainingOrderByGradeDesc(@Param("name") String name);

// すべての機能が一つのリポジトリで利用可能：
// 1. JpaRepository標準機能（save, findById, delete等）
// 2. JpaSpecificationExecutor動的検索
// 3. カスタム複雑クエリ（findStudentsWithComplexCriteria等）
}
🔗 Service層での活用パターン
@Service
@Transactional(readOnly = true)
public class StudentSearchServiceImpl implements StudentSearchService {

@Autowired
private StudentRepository studentRepository;

/**
* 標準的な動的検索（Specification活用）
*/
public List<StudentDto> searchStudents(StudentSearchCriteria criteria) {
Specification<Student> spec = StudentSpecifications.searchByCriteria(criteria);
return studentRepository.findAll(spec).stream()
.map(this::convertToDto)
.collect(Collectors.toList());
}

/**
* 複雑な条件検索（カスタムリポジトリ活用）
*/
public List<StudentDto> searchStudentsAdvanced(StudentSearchCriteria criteria) {
return studentRepository.findStudentsWithComplexCriteria(criteria).stream()
.map(this::convertToDto)
.collect(Collectors.toList());
}

/**
* 統計データ取得
*/
public List<MajorStatisticsDto> getMajorStatistics() {
return studentRepository.getGradeStatisticsByMajor().stream()
.map(this::convertToStatisticsDto)
.collect(Collectors.toList());
}
}
